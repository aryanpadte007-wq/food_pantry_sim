    B --> C[Create Environment,<br/>Pantry counter + stock]

    E -- No --> Z([Stop new arrivals])
    E -- Yes --> F[New student<br/>pick diet]
    F --> D
    F --> G{Volunteer free<br/>before max_wait?}

    G -- Yes --> I[Pack parcel]

    J -- No --> L{General stock left<br/>and diet not general?}

    L -- No --> N[outcome = no_food]

  1. 01.

    Basic queue

    Students arrive at random and wait for one volunteer.

    Generators · Resource · timeout

    View stage1_basic_queue.py
    """
Stage 1 - Basic queue.

Goal: learn the core SimPy idea. Students arrive at random times and
queue for ONE volunteer. We print what happens. No stock, no statistics.

Concepts shown: generator functions, simpy.Environment, simpy.Resource,
env.timeout, random arrival times.
"""

import random
import simpy

OPEN_MINUTES = 30       # short run so the output is readable
MEAN_ARRIVAL_GAP = 1.5  # average minutes between students
MEAN_SERVICE_TIME = 2.5 # average minutes to pack a parcel


def student(env, sid, counter):
    """One student arrives, waits for the volunteer, gets served, leaves."""
    arrived = env.now
    with counter.request() as req:
        yield req  # wait in the queue
        print(f"{env.now:6.1f}  student {sid} served after {env.now - arrived:.1f} min wait")
        yield env.timeout(random.expovariate(1 / MEAN_SERVICE_TIME))


def arrivals(env, counter):
    """Create new students until closing time."""
    sid = 0
    while env.now < OPEN_MINUTES:
        yield env.timeout(random.expovariate(1 / MEAN_ARRIVAL_GAP))
        sid += 1
        env.process(student(env, sid, counter))


if __name__ == "__main__":
    random.seed(42)
    env = simpy.Environment()
    counter = simpy.Resource(env, capacity=1)  # one volunteer
    env.process(arrivals(env, counter))
    env.run(until=OPEN_MINUTES + 30)

    """
Stage 2 - Settings in one place + basic statistics.

Changes from Stage 1:
  * A Config dataclass holds every setting, so scenarios are easy to change.
  * A Pantry class keeps the counter and a list of records.
  * Students give up if they wait longer than max_wait.
  * We report average wait and % who gave up instead of printing every event.

New concepts: dataclasses, classes, lists of dicts, the `|` (either-or) event.
"""

import random
import statistics
from dataclasses import dataclass

import simpy


@dataclass
class Config:
    """Settings for one run (minutes)."""
    volunteers: int = 1
    open_minutes: int = 180
    mean_arrival_gap: float = 1.5
    mean_service_time: float = 2.5
    max_wait: float = 20.0
    seed: int = 42


class Pantry:
    """Counter (volunteers) plus a list of what happened to each student."""

    def __init__(self, env, cfg):
        self.env = env
        self.cfg = cfg
        self.counter = simpy.Resource(env, capacity=cfg.volunteers)
        self.records = []


def student(env, sid, pantry):
    arrived = env.now
    with pantry.counter.request() as req:
        result = yield req | env.timeout(pantry.cfg.max_wait)
        waited = env.now - arrived
        if req in result:
            yield env.timeout(random.expovariate(1 / pantry.cfg.mean_service_time))
            outcome = "served"
        else:
            outcome = "gave_up"
    pantry.records.append({"student": sid, "wait": waited, "outcome": outcome})


def arrivals(env, pantry):
    sid = 0
    while env.now < pantry.cfg.open_minutes:
        yield env.timeout(random.expovariate(1 / pantry.cfg.mean_arrival_gap))
        sid += 1
        env.process(student(env, sid, pantry))


def run(cfg):
    random.seed(cfg.seed)
    env = simpy.Environment()
    pantry = Pantry(env, cfg)
    env.process(arrivals(env, pantry))
    env.run(until=cfg.open_minutes + cfg.max_wait + 30)
    return pantry.records


if __name__ == "__main__":
    for v in (1, 2, 3):
        recs = run(Config(volunteers=v))
        gave_up = sum(r["outcome"] == "gave_up" for r in recs)
        avg_wait = statistics.mean(r["wait"] for r in recs)
        print(f"{v} volunteer(s): {len(recs)} students, "
              f"avg wait {avg_wait:.1f} min, gave up {100*gave_up/len(recs):.1f}%")
  3. 03.

    Food stock & diets

    One Container per diet: suitable, unsuitable, no food, gave up.

    Container · dict · weighted choice

    View stage3_food_stock.py
    """
Stage 3 - Food stock and culturally appropriate food.

Changes from Stage 2:
  * Each student has a dietary need (general / vegetarian / halal).
  * Stock is held in one simpy.Container per diet group.
  * Outcomes are now: suitable, unsuitable (got general food instead),
    no_food (all stock gone), gave_up.

Why: research on international students in Australia (Leung et al., 2025)
found a lack of culturally appropriate food is a key barrier. This stage
makes that visible in the model. Diet shares are ASSUMPTIONS.

New concepts: simpy.Container, dictionaries, random.choices with weights,
sub-processes that return a value.
"""

import random
from collections import Counter
from dataclasses import dataclass, field

import simpy

DIET_SHARES = {"general": 0.55, "vegetarian": 0.25, "halal": 0.20}


@dataclass
class Config:
    volunteers: int = 2
    open_minutes: int = 180
    mean_arrival_gap: float = 1.5
    mean_service_time: float = 2.5
    max_wait: float = 20.0
    stock: dict = field(default_factory=lambda: {"general": 70, "vegetarian": 25, "halal": 15})
    seed: int = 42


class Pantry:
    def __init__(self, env, cfg):
        self.env = env
        self.cfg = cfg
        self.counter = simpy.Resource(env, capacity=cfg.volunteers)
        self.stock = {d: simpy.Container(env, init=n, capacity=max(n, 1))
                      for d, n in cfg.stock.items()}
        self.records = []

    def serve(self, diet):
        """Pack a parcel and report whether it met the student's need."""
        yield self.env.timeout(random.expovariate(1 / self.cfg.mean_service_time))
        if self.stock[diet].level > 0:
            yield self.stock[diet].get(1)
            return "suitable"
        if diet != "general" and self.stock["general"].level > 0:
            yield self.stock["general"].get(1)
            return "unsuitable"
        return "no_food"


def student(env, sid, pantry):
    arrived = env.now
    diet = random.choices(list(DIET_SHARES), weights=DIET_SHARES.values())[0]
    with pantry.counter.request() as req:
        result = yield req | env.timeout(pantry.cfg.max_wait)
        waited = env.now - arrived
        outcome = (yield env.process(pantry.serve(diet))) if req in result else "gave_up"
    pantry.records.append({"student": sid, "diet": diet, "wait": waited, "outcome": outcome})


def arrivals(env, pantry):
    sid = 0
    while env.now < pantry.cfg.open_minutes:
        yield env.timeout(random.expovariate(1 / pantry.cfg.mean_arrival_gap))
        sid += 1
        env.process(student(env, sid, pantry))


def run(cfg):
    random.seed(cfg.seed)
    env = simpy.Environment()
    pantry = Pantry(env, cfg)
    env.process(arrivals(env, pantry))
    env.run(until=cfg.open_minutes + cfg.max_wait + 30)
    return pantry.records


if __name__ == "__main__":
    recs = run(Config())
    print("Overall:", dict(Counter(r["outcome"] for r in recs)))
    for diet in DIET_SHARES:
        print(f"  {diet:<11}", dict(Counter(r["outcome"] for r in recs if r["diet"] == diet)))

    """
Stage 5 - Command-line options, CSV + chart export.

A small SimPy model of a student-run campus food bank, modelled on the
Swinburne Student Union (SSU) FoodBank, a free walk-in service open
10am-1pm on weekdays (180 minutes, as used below).

Why this matters:
    Australian research reports food insecurity among university students,
    and a 2025 study of international students (Leung et al., Health
    Promotion Journal of Australia) found a lack of culturally appropriate
    food and inefficient pantry operations were key barriers. This program
    lets a food bank coordinator test simple choices - volunteers, stock and
    culturally suitable stock - and see how many students get a parcel
    that meets their needs.

What it is NOT:
    A real forecasting tool. All numbers below are illustrative assumptions
    and should be replaced with real pantry data before drawing conclusions.

Run:
    python pantry_sim.py                 # compare built-in scenarios
    python pantry_sim.py --volunteers 3  # run one custom scenario
"""

import argparse
import random
import statistics
from dataclasses import dataclass, replace

import pandas as pd
import simpy

# Dietary groups a student may need. Shares are ASSUMPTIONS for illustration.
DIET_SHARES = {"general": 0.55, "vegetarian": 0.25, "halal": 0.20}


@dataclass
class Config:
    """All settings for one simulation run (times are in minutes)."""

    name: str = "Baseline"
    volunteers: int = 2            # people serving at the counter
    open_minutes: int = 180        # pantry open for 3 hours
    mean_arrival_gap: float = 1.5  # average minutes between student arrivals
    mean_service_time: float = 2.5  # average minutes to pack one parcel
    max_wait: float = 20.0         # students give up after waiting this long
    stock: dict = None             # starting parcels per diet group
    seed: int = 42

    def __post_init__(self):
        if self.stock is None:
            self.stock = {"general": 70, "vegetarian": 25, "halal": 15}


class Pantry:
    """The pantry: a counter (volunteers) plus food stock per diet group."""

    def __init__(self, env, cfg):
        self.env = env
        self.cfg = cfg
        self.counter = simpy.Resource(env, capacity=cfg.volunteers)
        # One SimPy Container per diet group, so each can run out separately.
        self.stock = {
            diet: simpy.Container(env, init=amount, capacity=max(amount, 1))
            for diet, amount in cfg.stock.items()
        }
        self.records = []  # one dict per student, filled during the run

    def serve(self, diet):
        """Pack a parcel. Returns the outcome as a short string."""
        yield self.env.timeout(random.expovariate(1 / self.cfg.mean_service_time))
        if self.stock[diet].level > 0:
            yield self.stock[diet].get(1)
            return "suitable"
        if diet != "general" and self.stock["general"].level > 0:
            # Something was given, but it may not meet their dietary need.
            yield self.stock["general"].get(1)
            return "unsuitable"
        return "no_food"


def pick_diet():
    """Randomly choose a diet group using DIET_SHARES."""
    return random.choices(list(DIET_SHARES), weights=DIET_SHARES.values())[0]


def student(env, sid, pantry):
    """One student: arrive, queue (or give up), get a parcel, leave."""
    arrived = env.now
    diet = pick_diet()
    with pantry.counter.request() as req:
        # Wait for a volunteer OR run out of patience - whichever comes first.
        result = yield req | env.timeout(pantry.cfg.max_wait)
        waited = env.now - arrived
        if req not in result:
            outcome = "gave_up"
        else:
            outcome = yield env.process(pantry.serve(diet))
    pantry.records.append(
        {"student": sid, "diet": diet, "wait": round(waited, 2), "outcome": outcome}
    )


def arrivals(env, pantry):
    """Generate students until closing time."""
    sid = 0
    while env.now < pantry.cfg.open_minutes:
        yield env.timeout(random.expovariate(1 / pantry.cfg.mean_arrival_gap))
        sid += 1
        env.process(student(env, sid, pantry))


def run(cfg):
    """Run one scenario and return a DataFrame of student records."""
    random.seed(cfg.seed)
    env = simpy.Environment()
    pantry = Pantry(env, cfg)
    env.process(arrivals(env, pantry))
    # Let students already in the queue finish after the doors close.
    env.run(until=cfg.open_minutes + cfg.max_wait + 30)
    return pd.DataFrame(pantry.records)


def summarise(cfg, df):
    """Turn raw records into the key numbers a coordinator cares about."""
    total = len(df)
    share = lambda o: round(100 * (df["outcome"] == o).sum() / total, 1) if total else 0.0
    served = df[df["outcome"] != "gave_up"]
    return {
        "scenario": cfg.name,
        "volunteers": cfg.volunteers,
        "students": total,
        "suitable_%": share("suitable"),
        "unsuitable_%": share("unsuitable"),
        "no_food_%": share("no_food"),
        "gave_up_%": share("gave_up"),
        "avg_wait_min": round(statistics.mean(served["wait"]), 1) if len(served) else 0.0,
    }


def run_replications(cfg, reps=20):
    """Repeat a scenario with different seeds and average the results.

    One random run can be lucky or unlucky; averaging gives a fairer view.
    """
    rows = [summarise(cfg, run(replace(cfg, seed=cfg.seed + i))) for i in range(reps)]
    avg = pd.DataFrame(rows).drop(columns="scenario").mean().round(1).to_dict()
    return {"scenario": cfg.name, **avg}


def default_scenarios():
    """A few 'what if' settings a pantry coordinator might test."""
    base = Config()
    return [
        base,
        replace(base, name="+1 volunteer", volunteers=3),
        replace(base, name="Same stock, culturally matched",
                stock={"general": 60, "vegetarian": 28, "halal": 22}),
        replace(base, name="+1 volunteer & matched stock", volunteers=3,
                stock={"general": 60, "vegetarian": 28, "halal": 22}),
    ]


def plot(results, path="results.png"):
    """Save a stacked bar chart of outcomes per scenario."""
    import matplotlib
    matplotlib.use("Agg")
    import matplotlib.pyplot as plt

    cols = ["suitable_%", "unsuitable_%", "no_food_%", "gave_up_%"]
    ax = results.set_index("scenario")[cols].plot(
        kind="barh", stacked=True, figsize=(9, 4),
        color=["#2a9d8f", "#e9c46a", "#e76f51", "#8d99ae"])
    ax.set_xlabel("% of students")
    ax.set_title("Pantry outcomes by scenario (average of runs)")
    ax.invert_yaxis()
    plt.tight_layout()
    plt.savefig(path, dpi=120)
    plt.close()
    return path


def parse_args(argv=None):
    """Read command-line options for a custom scenario."""
    p = argparse.ArgumentParser(description="Campus food pantry simulator")
    p.add_argument("--volunteers", type=int, help="run one custom scenario")
    p.add_argument("--minutes", type=int, default=180, help="opening time")
    p.add_argument("--reps", type=int, default=20, help="runs per scenario")
    p.add_argument("--csv", default="results.csv", help="where to save results")
    return p.parse_args(argv)


def main(argv=None):
    """Entry point: run scenarios, print a table, save CSV and chart."""
    args = parse_args(argv)
    if args.volunteers is not None:
        if args.volunteers < 1:
            raise SystemExit("Need at least 1 volunteer.")
        scenarios = [Config(name="Custom", volunteers=args.volunteers,
                            open_minutes=args.minutes)]
    else:
        scenarios = [replace(s, open_minutes=args.minutes) for s in default_scenarios()]

    results = pd.DataFrame([run_replications(s, args.reps) for s in scenarios])
    print(results.to_string(index=False))
    results.to_csv(args.csv, index=False)
    print(f"\nSaved {args.csv} and {plot(results)}")
    print("Note: inputs are illustrative assumptions, not real pantry data.")
    return results


if __name__ == "__main__":
    main()

    """
Campus Food Pantry Simulator (final version - Stage 6).

A small SimPy model of a student-run campus food bank, modelled on the
Swinburne Student Union (SSU) FoodBank, a free walk-in service open
10am-1pm on weekdays (180 minutes, as used below).

Why this matters:

    (Kent et al., 2025). A study of international students (Leung et al.,

What it is NOT:

Run:
    python pantry_sim.py                          # built-in scenarios

"""

import argparse
import csv

import random
import statistics
import time
from dataclasses import dataclass, field, replace

import pandas as pd
import simpy

log = logging.getLogger("pantry")

# Dietary groups and the share of students needing each (ASSUMPTIONS).
DIET_SHARES = {"general": 0.55, "vegetarian": 0.25, "halal": 0.20}
OUTCOMES = ("suitable", "unsuitable", "no_food", "gave_up")


def default_stock():
    """Starting parcels per diet group (a fresh dict for every Config)."""

class Pantry:
    """The pantry: a counter (volunteers) plus food stock per diet group."""

    def __init__(self, env, cfg):
        self.env = env
        self.cfg = cfg
        self.counter = simpy.Resource(env, capacity=cfg.volunteers)
        # One Container per diet group, so each can run out separately.
        self.stock = {
            diet: simpy.Container(env, init=amount, capacity=max(amount, 1))
            for diet, amount in cfg.stock.items()
        }
        self.records = []  # one dict per student

    def serve(self, diet):
        """Pack a parcel and return how well it met the student's need."""
        yield self.env.timeout(random.expovariate(1 / self.cfg.mean_service_time))
        if self.stock[diet].level > 0:
            yield self.stock[diet].get(1)
            return "suitable"
        if diet != "general" and self.stock["general"].level > 0:
            yield self.stock["general"].get(1)  # food, but not what they need
            return "unsuitable"
        return "no_food"


def pick_diet():
    """Randomly choose a diet group using DIET_SHARES as weights."""
    return random.choices(list(DIET_SHARES), weights=DIET_SHARES.values())[0]


def student(env, sid, pantry):
    """One student: arrive, queue (or give up), get a parcel, leave."""
    arrived = env.now
    diet = pick_diet()
    with pantry.counter.request() as req:
        # Wait for a volunteer OR run out of patience, whichever is first.
        result = yield req | env.timeout(pantry.cfg.max_wait)
        waited = env.now - arrived
        if req in result:
            outcome = yield env.process(pantry.serve(diet))
        else:
            outcome = "gave_up"
    log.debug("t=%6.1f student %3d %-10s %-10s waited %.1f",
              env.now, sid, diet, outcome, waited)
    pantry.records.append(
        {"student": sid, "diet": diet, "wait": round(waited, 2), "outcome": outcome})


def arrivals(env, pantry):
    """Generate students until closing time."""
    sid = 0
    while True:
        yield env.timeout(random.expovariate(1 / pantry.cfg.mean_arrival_gap))
        if env.now >= pantry.cfg.open_minutes:
            break  # doors closed - this student arrived too late
        sid += 1
        env.process(student(env, sid, pantry))

    env = simpy.Environment()

    env.run(until=cfg.open_minutes + cfg.max_wait + 30)  # let the queue clear
    return pd.DataFrame(pantry.records, columns=["student", "diet", "wait", "outcome"])

    showing a confidence interval, tells us how much to trust a difference.

"""
Campus FoodBank Simulator (final version - Stage 8, simplified).

A SimPy model of a free, walk-in campus food bank, based on the
Swinburne Student Union FoodBank (open 10am-1pm on weekdays).

Why: food insecurity among Australian university students rose from 42%
to 53% between 2022 and 2024, with international students at higher risk
(Kent et al., 2025), and students report a lack of culturally appropriate
food at campus pantries (Leung et al., 2025). The model asks: how many
students leave with food that suits their diet, not just any food?

All operating numbers are illustrative assumptions, not real data.

Run:   python pantry_sim.py                    (reads scenarios.csv)
       python pantry_sim.py my.csv --reps 50
"""

import argparse
import csv
import random
import statistics
from collections import Counter
from dataclasses import dataclass, field, replace

import simpy

# Share of students needing each diet (ASSUMPTION).
DIET_SHARES = {"general": 0.55, "vegetarian": 0.25, "halal": 0.20}
OUTCOMES = ("suitable", "unsuitable", "no_food", "gave_up")


@dataclass
class Config:
    """Settings for one scenario. Times are in minutes."""

    name: str = "Baseline"
    volunteers: int = 2
    stock: dict = field(default_factory=lambda: {"general": 70, "vegetarian": 25, "halal": 15})
    open_minutes: float = 180       # 10am-1pm
    arrival_gap: float = 1.5        # average minutes between students
    service_time: float = 2.5       # average minutes to pack a parcel
    max_wait: float = 20            # students give up after this long
    seed: int = 42

    def validate(self):
        """Raise ValueError if a setting makes no sense; return self."""
        if self.volunteers < 1:
            raise ValueError(f"{self.name}: need at least 1 volunteer")
        if set(self.stock) != set(DIET_SHARES):
            raise ValueError(f"{self.name}: stock needs {list(DIET_SHARES)}")
        if min(self.stock.values()) < 0:
            raise ValueError(f"{self.name}: stock cannot be negative")
        return self


def choose_parcel(stock, diet):
    """Return (outcome, diet to take stock from, or None).

    The core rule, kept free of SimPy so it is easy to test and change.
    """
    if stock[diet] > 0:
        return "suitable", diet
    if diet != "general" and stock["general"] > 0:
        return "unsuitable", "general"      # food, but not what they need
    return "no_food", None


def run(cfg):
    """Simulate one session. Return a list of (diet, wait, outcome)."""
    rng = random.Random(cfg.seed)
    env = simpy.Environment()
    volunteers = simpy.Resource(env, capacity=cfg.volunteers)
    stock = dict(cfg.stock)                 # copy, so cfg is not changed
    records = []

    def student(diet):
        arrived = env.now
        with volunteers.request() as turn:
            # Wait for a volunteer OR give up - whichever happens first.
            result = yield turn | env.timeout(cfg.max_wait)
            wait = env.now - arrived
            if turn not in result:
                outcome = "gave_up"
            else:
                yield env.timeout(rng.expovariate(1 / cfg.service_time))
                outcome, take = choose_parcel(stock, diet)
                if take:
                    stock[take] -= 1
        records.append((diet, wait, outcome))

    def arrivals():
        while True:
            yield env.timeout(rng.expovariate(1 / cfg.arrival_gap))
            if env.now >= cfg.open_minutes:
                return                      # doors closed
            diet = rng.choices(list(DIET_SHARES), weights=DIET_SHARES.values())[0]
            env.process(student(diet))

    env.process(arrivals())
    env.run()                               # ends when the last student leaves
    return records


def summarise(records):
    """Percentage of students per outcome, plus average wait of those served."""
    total = len(records) or 1
    counts = Counter(outcome for _, _, outcome in records)
    row = {f"{o}_%": 100 * counts[o] / total for o in OUTCOMES}
    waits = [w for _, w, o in records if o != "gave_up"]
    row["wait_min"] = statistics.fmean(waits) if waits else 0.0
    return row


def replicate(cfg, reps=20):
    """Run a scenario `reps` times (seeds seed, seed+1, ...) and average.

    Also returns a 95% confidence interval for suitable %, so we can tell
    a real difference from random noise.
    """
    rows = [summarise(run(replace(cfg, seed=cfg.seed + i))) for i in range(reps)]
    result = {"scenario": cfg.name, "volunteers": cfg.volunteers}
    for key in rows[0]:
        result[key] = round(statistics.fmean(r[key] for r in rows), 1)
    suitable = [r["suitable_%"] for r in rows]
    result["ci95"] = round(1.96 * statistics.stdev(suitable) / reps ** 0.5, 1) if reps > 1 else 0.0
    return result

Likely questions.

The full set of 43 is in docs/QA.md.

Is this real data?

The setting is real (SSU FoodBank hours, walk-in). The operating numbers are labelled assumptions, tested across a range in experiments.py.

Is the main difference real or noise?

84.0 ± 3.2 vs 89.0 ± 2.9 over 20 runs. Compared seed by seed, matched stock wins 16 of 20 runs by 5.0 ± 1.5 points, so the gain is likely real.

Where is the Indigenous perspective?

The project does not model Indigenous communities. Any adaptation for Aboriginal or Torres Strait Islander students should be led by that community through Swinburne's Moondani Toombadool Centre and follow CARE and Maiam nayri Wingara principles.

What bug did testing find?

The arrivals loop checked closing time before waiting, so one student always arrived after closing. A test caught it; the fix moved the check after the wait.

What does req | env.timeout(max_wait) do?

The student waits for a volunteer or until patience runs out, whichever happens first.

Why one Container per diet?

So halal and vegetarian stock can run out on their own. One total would hide the problem the project is about.
